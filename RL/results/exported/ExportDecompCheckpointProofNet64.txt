theorem exercise_4_4a
  {α : Type} [MetricSpace α]
  {β : Type} [MetricSpace β]
  (f : α → β)
  (s : Set α)
  (h₁ : Continuous f)
  (h₂ : Dense s)
  : f '' Set.univ ⊆ closure (f '' s) :=
by
  have h₃ : f '' Set.univ ⊆ closure (f '' s) := by
    -- For any point y in the image of f over the entire space
    rintro y ⟨x, hx, rfl⟩
    -- Since s is dense in α, for any x, there exists a sequence in s converging to x
    have : x ∈ closure s := h₂ x
    -- The image of a convergent sequence under a continuous function is a convergent sequence
    refine' mem_closure_image h₁.continuousAt this
  exact h₃
